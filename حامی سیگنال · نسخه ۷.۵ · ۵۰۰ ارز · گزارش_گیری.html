import asyncio
import ccxt.async_support as ccxt
import logging

logging.basicConfig(level=logging.INFO, format='%(asctime)s - [%(levelname)s] - %(message)s')

class SuperstarTraderCore:
    def __init__(self):
        self.exchanges = {
            'mexc': ccxt.mexc({'enableRateLimit': True}),
            'binance': ccxt.binance({'enableRateLimit': True}),
            'coinex': ccxt.coinex({'enableRateLimit': True}),
            'bybit': ccxt.bybit({'enableRateLimit': True})
        }
        self.risk_settings = {
            'max_risk_per_trade_percent': 2.0,
            'default_leverage': 5,
            'trailing_stop_callback': 0.01
        }

    async def close_exchanges(self):
        for exchange in self.exchanges.values():
            await exchange.close()

    async def fetch_ticker_price(self, exchange_name, symbol):
        exchange = self.exchanges.get(exchange_name)
        if not exchange:
            logging.error(f"صرافی {exchange_name} تعریف نشده است.")
            return None
        try:
            ticker = await exchange.fetch_ticker(symbol)
            return ticker['last']
        except Exception as e:
            logging.error(f"خطا در دریافت قیمت از {exchange_name} برای {symbol}: {e}")
            return None

    async def compare_prices(self, symbol):
        tasks = [self.fetch_ticker_price(ex_name, symbol) for ex_name in self.exchanges.keys()]
        prices = await asyncio.gather(*tasks)
        market_data = {}
        for ex_name, price in zip(self.exchanges.keys(), prices):
            if price:
                market_data[ex_name] = price
        return market_data

    def calculate_position_size(self, account_balance, entry_price, stop_loss_price):
        risk_amount = account_balance * (self.risk_settings['max_risk_per_trade_percent'] / 100)
        risk_per_unit = abs(entry_price - stop_loss_price)
        if risk_per_unit == 0:
            return 0
        position_size = risk_amount / risk_per_unit
        return position_size

    async def execute_smart_order(self, exchange_name, symbol, order_type, side, amount, price=None):
        exchange = self.exchanges.get(exchange_name)
        if not exchange:
            return {"status": "error", "message": "صرافی نامعتبر است."}
        try:
            await exchange.load_markets()
            if order_type == 'market':
                order = await exchange.create_market_order(symbol, side, amount)
            elif order_type == 'limit':
                order = await exchange.create_limit_order(symbol, side, amount, price)
            else:
                return {"status": "error", "message": "نوع سفارش ناشناخته است."}
            logging.info(f"سفارش موفق در {exchange_name}: {order.get('id')}")
            return {"status": "success", "order": order}
        except ccxt.InsufficientFunds as e:
            logging.error(f"موجودی ناکافی: {e}")
            return {"status": "error", "message": "موجودی حساب برای این حجم معامله کافی نیست."}
        except Exception as e:
            logging.error(f"خطای سیستمی در ثبت سفارش: {e}")
            return {"status": "error", "message": str(e)}

async def main():
    trader = SuperstarTraderCore()
    symbol = 'BTC/USDT'
    print("--- 🚀 سامانه حامی سیگنال (نسخه حرفه‌ای CCXT) ---")

if __name__ == "__main__":
